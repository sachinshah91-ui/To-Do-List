Setting up Google Sheets

Follow these steps to connect your Google Sheet as a live database. Takes about 5 minutes.

Step 1 — Create the Sheet

Go to sheets.google.com and create a new spreadsheet. Name it TaskTracker.

Step 2 — Add headers in Row 1

In the first sheet, add these exact headers in columns A–H:

id | title | description | status | billing_status | payment_status | created_at | updated_at
Step 3 — Open Apps Script

In your Sheet, click Extensions → Apps Script. Delete any existing code and paste the script below.

Step 4 — Paste this Apps Script code
const SHEET_NAME = 'Sheet1';

function doGet(e) {
  const action = e.parameter.action;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);
  if (action === 'getAll') {
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    const rows = data.slice(1).map(r => {
      const obj = {};
      headers.forEach((h,i) => obj[h] = r[i]);
      return obj;
    });
    return jsonResp(rows);
  }
  return jsonResp({ error: 'Unknown action' });
}

function doPost(e) {
  const body = JSON.parse(e.postData.contents);
  const { action, task } = body;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);
  const headers = ['id','title','description','status','billing_status','payment_status','created_at','updated_at'];

  if (action === 'create') {
    sheet.appendRow(headers.map(h => task[h] || ''));
    return jsonResp({ ok: true, task });
  }
  if (action === 'update') {
    const data = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === task.id) {
        headers.forEach((h, j) => sheet.getRange(i+1, j+1).setValue(task[h] !== undefined ? task[h] : data[i][j]));
        return jsonResp({ ok: true });
      }
    }
    return jsonResp({ ok: false, error: 'Not found' });
  }
  if (action === 'delete') {
    const data = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === task.id) {
        sheet.deleteRow(i + 1);
        return jsonResp({ ok: true });
      }
    }
    return jsonResp({ ok: false, error: 'Not found' });
  }
  return jsonResp({ error: 'Unknown action' });
}

function jsonResp(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
Step 5 — Deploy as Web App

Click Deploy → New deployment. Choose type Web app. Set Execute as: Me and Who has access: Anyone. Click Deploy, authorize, then copy the Web App URL.

Step 6 — Paste URL in TaskTracker

Close this guide and paste the URL into the connection box. Click Connect. Done — your tasks will sync across all devices!
