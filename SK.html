// ── PASTE THIS INTO YOUR APPS SCRIPT (replaces previous code) ── // Sheet headers in Row 1 must be exactly: // id | title | description | status | billing_status | payment_status | priority | due_date | tags | created_at | updated_at const SHEET_NAME = 'Sheet1'; function doGet(e) { const action = e.parameter.action; const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); if (action === 'getAll') { const data = sheet.getDataRange().getValues(); const headers = data[0]; const rows = data.slice(1) .filter(r => r[0] !== '') .map(r => { const obj = {}; headers.forEach((h, i) => obj[h] = r[i]); return obj; }); return jsonResp(rows); } return jsonResp({ error: 'Unknown action' }); } function doPost(e) { const body = JSON.parse(e.postData.contents); const { action, task } = body; const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); const headers = [ 'id','title','description','status','billing_status', 'payment_status','priority','due_date','tags','created_at','updated_at' ]; if (action === 'create') { sheet.appendRow(headers.map(h => task[h] !== undefined ? task[h] : '')); return jsonResp({ ok: true }); } if (action === 'update') { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(task.id)) { headers.forEach((h, j) => { if (task[h] !== undefined) sheet.getRange(i + 1, j + 1).setValue(task[h]); }); return jsonResp({ ok: true }); } } return jsonResp({ ok: false, error: 'Not found' }); } if (action === 'delete') { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(task.id)) { sheet.deleteRow(i + 1); return jsonResp({ ok: true }); } } return jsonResp({ ok: false, error: 'Not found' }); } return jsonResp({ error: 'Unknown action' }); } function jsonResp(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
